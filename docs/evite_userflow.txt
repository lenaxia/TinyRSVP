        let currentDate = new Date(2026, 0, 1);
        let selectedDate = null;

        function renderCalendar() {
            const grid = document.getElementById('calendarGrid');
            const monthDisplay = document.getElementById('currentMonth');
            
            const dayHeaders = Array.from(grid.children).slice(0, 7);
            grid.innerHTML = '';
            dayHeaders.forEach(header => grid.appendChild(header));

            const year = currentDate.getFullYear();
            const month = currentDate.getMonth();
            
            monthDisplay.textContent = currentDate.toLocaleDateString('en-US', { month: 'long', year: 'numeric' });

            const firstDay = new Date(year, month, 1).getDay();
            const daysInMonth = new Date(year, month + 1, 0).getDate();
            const daysInPrevMonth = new Date(year, month, 0).getDate();

            for (let i = firstDay - 1; i >= 0; i--) {
                const day = document.createElement('div');
                day.className = 'calendar-day other-month';
                day.textContent = daysInPrevMonth - i;
                grid.appendChild(day);
            }

            for (let i = 1; i <= daysInMonth; i++) {
                const day = document.createElement('div');
                day.className = 'calendar-day';
                day.textContent = i;
                
                const dayDate = new Date(year, month, i);
                const today = new Date();
                if (dayDate.toDateString() === today.toDateString()) {
                    day.classList.add('today');
                }
                
                day.onclick = () => selectDate(dayDate, day);
                grid.appendChild(day);
            }

            const totalCells = grid.children.length - 7;
            const remainingCells = 42 - totalCells;
            for (let i = 1; i <= remainingCells; i++) {
                const day = document.createElement('div');
                day.className = 'calendar-day other-month';
                day.textContent = i;
                grid.appendChild(day);
            }
        }

        function selectDate(date, element) {
            document.querySelectorAll('.calendar-day').forEach(d => d.classList.remove('selected'));
            element.classList.add('selected');
            selectedDate = date;
        }

        function changeMonth(delta) {
            currentDate.setMonth(currentDate.getMonth() + delta);
            renderCalendar();
        }

        function openModal(modalId) {
            document.getElementById(modalId).classList.add('active');
            if (modalId === 'dateModal') {
                renderCalendar();
            }
        }

        function closeModal(modalId) {
            document.getElementById(modalId).classList.remove('active');
        }

        function updateCounter(id, delta) {
            const element = document.getElementById(id);
            let value = parseInt(element.textContent);
            value = Math.max(0, Math.min(9, value + delta));
            element.textContent = value;
        }

        function addBringItem() {
            const list = document.getElementById('bringList');
            const item = document.createElement('div');
            item.className = 'list-item';
            item.innerHTML = `
                <input type="text" placeholder="Item name">
                <input type="number" placeholder="Qty" style="width: 80px;">
                <button class="remove-btn" onclick="removeItem(this)">×</button>
            `;
            list.appendChild(item);
        }

        function removeItem(button) {
            button.parentElement.remove();
        }

        renderCalendar();
    </script>

Event Creation Flow Overview

The event creation system provides hosts with a streamlined wizard-based interface for configuring and managing event invitations. The flow is organized into three primary stages: Design, Details, and Review, with a persistent progress indicator allowing users to track their position within the creation process.

Core Event Configuration

The initial configuration phase captures fundamental event information through a series of expandable form sections. Hosts begin by defining the event title, which serves as the primary identifier for the invitation. The date and time selection interface presents an integrated calendar component with month-based navigation, allowing hosts to select specific dates through an interactive grid view. Time selection is facilitated through a dropdown menu offering 15-minute interval options throughout a 24-hour period, accompanied by timezone specification to ensure clarity for guests across different regions.

Location specification accepts freeform text input, providing flexibility for physical addresses, virtual meeting links, or descriptive location information. The system accommodates both traditional venue-based events and remote gatherings without imposing structural constraints on how location data is entered.

Host Management and Collaboration

The host details section captures primary host information including name and optional contact details such as phone numbers. The system supports collaborative hosting through a co-host invitation mechanism, enabling multiple individuals to share hosting responsibilities and potentially access event management capabilities. This feature recognizes that many events involve coordination between multiple organizers.

Guest Management and RSVP Configuration

The RSVP options section provides granular control over guest response mechanisms and list visibility. Hosts can enable or disable "Maybe" responses, offering flexibility in how guests indicate their attendance uncertainty. The private guest list toggle controls whether attendees can view other invited guests, addressing privacy preferences for different event types.

Family headcount tracking allows hosts to capture attendance numbers for guests bringing multiple family members, which is particularly relevant for events with capacity constraints or catering requirements. The plus-one configuration uses a counter interface allowing hosts to specify how many additional guests each invitee may bring, with a maximum of nine additional guests per invitation.

Guest capacity limits can be established to manage event size, automatically tracking total attendance against the specified maximum. RSVP deadline configuration ensures hosts receive responses within their planning timeline.

Contribution Coordination

The "What to Bring" list builder enables hosts to coordinate guest contributions for potluck-style events or gatherings requiring specific items. Each list entry captures an item description and optional quantity specification, allowing hosts to communicate needs clearly while enabling guests to claim specific contributions. This feature streamlines the coordination typically handled through separate communication channels.

Additional Engagement Features

The system supports designation of a guest of honor, appropriate for celebrations centered around specific individuals. Photo sharing capabilities include both host-managed galleries and collaborative guest albums, facilitating visual documentation and sharing before, during, and after the event.

Polling functionality allows hosts to gather guest input on various aspects such as preferred dates, menu options, or activity preferences. This democratic approach to event planning helps ensure the gathering meets attendee preferences.

External resource integration permits hosts to attach up to three external links, accommodating liability waivers, registration forms, or supplementary information hosted on other platforms.

Progressive Disclosure and Editing

The interface employs progressive disclosure principles, presenting core information in collapsed form with edit actions that reveal detailed configuration options through modal overlays. This approach maintains a clean primary interface while providing access to comprehensive settings when needed. Each section can be independently edited without disrupting other configured elements, supporting iterative refinement of event details.

The wizard structure guides hosts through logical configuration stages while maintaining flexibility to revisit and modify earlier sections. Draft saving functionality ensures work is preserved throughout the creation process, accommodating hosts who may need to complete configuration across multiple sessions.

This architecture balances simplicity for straightforward events with extensibility for complex gatherings requiring detailed coordination, guest contribution management, and collaborative planning features.
